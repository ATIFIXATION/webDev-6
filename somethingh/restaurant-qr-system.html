🗄️ Database Schemas (Mongoose)
// models/MenuItem.js
const menuItemSchema = new Schema({
  restaurantId: { type: ObjectId, ref: 'Restaurant', required: true },
  name:         { type: String, required: true },
  description:  String,
  price:        { type: Number, required: true },
  category:     { type: String, required: true },
  emoji:        String,
  isVeg:        { type: Boolean, default: true },
  isAvailable:  { type: Boolean, default: true },
  spiceLevel:   { type: String, enum: ['mild','medium','spicy'] },
}, { timestamps: true });

// models/Order.js
const orderSchema = new Schema({
  restaurantId: { type: ObjectId, ref: 'Restaurant' },
  tableId:      String,
  sessionId:    { type: ObjectId, ref: 'TableSession' },
  items: [{
    menuItemId: { type: ObjectId, ref: 'MenuItem' },
    name:       String,
    price:      Number,
    quantity:   Number,
  }],
  totalAmount:  Number,
  status: {
    type: String,
    enum: ['pending','preparing','served','paid'],
    default: 'pending'
  },
  specialNote:  String,
  orderId:      String,  // human-readable e.g. #ORD-001
}, { timestamps: true });

// models/TableSession.js
const tableSessionSchema = new Schema({
  restaurantId: { type: ObjectId, ref: 'Restaurant' },
  tableId:      String,
  isActive:     { type: Boolean, default: true },
  orders:       [{ type: ObjectId, ref: 'Order' }],
  startedAt:    { type: Date, default: Date.now },
  endedAt:      Date,
});
🔗 REST API Routes
GET    /api/menu/:restaurantId          — Public menu for customers
GET    /api/menu/:restaurantId/:id      — Single item details

POST   /api/menu                        — Add menu item (admin)
PUT    /api/menu/:id                    — Update item (admin)
DELETE /api/menu/:id                    — Delete item (admin)
PATCH  /api/menu/:id/toggle             — Toggle availability

POST   /api/orders                      — Place new order
GET    /api/orders/:restaurantId        — All orders (admin)
GET    /api/orders/status/:orderId      — Order status (customer)
PATCH  /api/orders/:id/status           — Update status (admin)

POST   /api/table/session               — Create table session
GET    /api/table/qr/:restaurantId      — Generate QR data

POST   /api/restaurant                  — Register restaurant
POST   /api/auth/login                  — Admin login → JWT

// Server → Client
socket.on('order-received', order)        — Admin gets new order
socket.on('order-status-update', data)   — Customer gets status
socket.on('order-confirmed', order)      — Customer confirmation

// server.js socket setup
io.on('connection', (socket) => {
  socket.on('join-restaurant', (id) => {
    socket.join(`restaurant:${id}`);
  });
  socket.on('new-order', async (data) => {
    const order = await Order.create(data);
    io.to(`restaurant:${data.restaurantId}`)
      .emit('order-received', order);
  });
});
🚀 Setup Instructions
# 1. Clone & install
git clone https://github.com/yourrepo/dineqr
cd dineqr/backend && npm install
cd ../frontend && npm install

# 2. Backend .env
MONGODB_URI=mongodb+srv://user:pass@cluster.mongodb.net/dineqr
JWT_SECRET=your_super_secret_key
PORT=5000
CLIENT_URL=http://localhost:5173

# 3. Run backend
cd backend && npm run dev   # nodemon server.js

# 4. Run frontend
cd frontend && npm run dev  # Vite dev server

📋 How QR Ordering Works
1. Customer scans QR code on table → Opens /menu?rid=RESTAURANT_ID&tid=TABLE_1
2. They browse the menu, add items to cart → Place order
3. Order appears instantly on Admin "Live Orders" panel via Socket.io
4. Admin updates status: Pending → Preparing → Served → Paid
5. Customer sees real-time status updates on their phone
